  GTM remains the only owner of completed-purchase conversion events.

  v4 reads the normal same-tab session context first and can also
  consume the optional same-origin localStorage backup written by
  the Nutrition Fundamentals sales page reliability patch.

<script>
(function(){
  "use strict";

  if (window.__NF_PURCHASE_RETURN_CONTEXT_V4__) return;
  window.__NF_PURCHASE_RETURN_CONTEXT_V4__ = true;

  var LEGACY_RAW_KEY = "nf_last_purchase_ctx";
  var BOOK_RAW_KEY = "nf.purchase.context.v1";
  var BOOK_BACKUP_KEY = "nf.purchase.context.backup.v1";
  var DISPLAY_KEY = "nf.thankyou.display.v4";
  var PREPARED_KEY = "nf.thankyou.prepared.v3";
  var FIRED_KEY = "nf_purchase_fired";

  var MAX_AGE_MS = 4 * 60 * 60 * 1000;
  var BOOK_SKU = "guide_nutrition_fundamentals_v1";
  var BOOK_PRICE = 34.99;
  var BOOK_NAME = "NeuForm Nutrition Fundamentals Digital Edition";

  function getSession(key){
    try { return sessionStorage.getItem(key); }
    catch (_) { return null; }
  }

  function setSession(key, value){
    try {
      sessionStorage.setItem(key, value);
      return true;
    } catch (_) {
      return false;
    }
  }

  function removeSession(key){
    try { sessionStorage.removeItem(key); }
    catch (_) {}
  }

  function getLocal(key){
    try { return localStorage.getItem(key); }
    catch (_) { return null; }
  }

  function removeLocal(key){
    try { localStorage.removeItem(key); }
    catch (_) {}
  }

  function parseJson(raw){
    if (!raw) return null;
    try {
      var value = JSON.parse(raw);
      return value && typeof value === "object" ? value : null;
    } catch (_) {
      return null;
    }
  }

  function safeSku(value){
    var sku = String(value || "").trim();
    return /^[A-Za-z0-9._-]{2,120}$/.test(sku) ? sku : "";
  }

  function safeText(value, limit){
    return String(value || "")
      .replace(/\s+/g, " ")
      .replace(/[<>{}"'`\\]/g, "")
      .trim()
      .slice(0, limit || 120);
  }

  function safePrice(value){
    var price = Number(value);
    if (!isFinite(price) || price <= 0 || price > 10000) return 0;
    return Math.round(price * 100) / 100;
  }

  function timestampIsFresh(timestamp, maxAge){
    var ts = Number(timestamp || 0);
    if (!ts) return false;
    var age = Date.now() - ts;
    return age >= -60000 && age <= (maxAge || MAX_AGE_MS);
  }

  function normalizeLegacy(value){
    if (!value || typeof value !== "object") return null;

    var sku = safeSku(value.sku);
    var price = safePrice(value.price);
    var timestamp = Number(value.ts || value.createdAt || 0);

    if (!sku || !price) return null;
    if (timestamp && !timestampIsFresh(timestamp, MAX_AGE_MS)) return null;

    return {
      version:4,
      ts:timestamp || Date.now(),
      sku:sku,
      price:price,
      productName:safeText(
        value.productName ||
        value.plan ||
        value.item_name ||
        "",
        120
      ),
      source:safeText(value.source || "purchase_return", 40),
      contextType:sku === BOOK_SKU ? "book" : "purchase"
    };
  }

  function normalizeBook(value){
    if (!value || typeof value !== "object") return null;
    if (safeSku(value.sku) !== BOOK_SKU) return null;

    var createdAt = Number(value.createdAt || value.ts || 0);
    var expiresAt = Number(value.expiresAt || 0);

    if (!timestampIsFresh(createdAt, MAX_AGE_MS)) return null;
    if (expiresAt && expiresAt <= Date.now()) return null;

    return {
      version:4,
      ts:createdAt,
      sku:BOOK_SKU,
      price:BOOK_PRICE,
      productName:BOOK_NAME,
      source:safeText(
        value.source || "nutrition_fundamentals_sales",
        40
      ),
      contextType:"book"
    };
  }

  function queryContext(){
    try {
      var params = new URLSearchParams(window.location.search || "");
      return normalizeLegacy({
        ts:Date.now(),
        sku:params.get("sku") || "",
        price:params.get("price") || "",
        productName:
          params.get("plan") ||
          params.get("item_name") ||
          "",
        source:"thankyou_query"
      });
    } catch (_) {
      return null;
    }
  }

  function returnFingerprint(context){
    var sessionId = "";

    try {
      sessionId = new URLSearchParams(
        window.location.search || ""
      ).get("session_id") || "";
    } catch (_) {}

    if (/^cs_(?:test_)?[A-Za-z0-9_]{12,220}$/.test(sessionId)) {
      return "session:" + sessionId;
    }

    return [
      "ctx",
      context.sku,
      String(context.price),
      String(context.ts || "")
    ].join(":");
  }

  function legacyPayload(context){
    return {
      version:4,
      ts:context.ts,
      sku:context.sku,
      price:context.price,
      plan:context.productName,
      productName:context.productName,
      source:context.source
    };
  }

  try {
    document.documentElement.classList.add(
      "nf-purchase-thankyou-page"
    );
  } catch (_) {}

  var sessionBookContext = normalizeBook(
    parseJson(getSession(BOOK_RAW_KEY))
  );

  var backupBookContext = sessionBookContext
    ? null
    : normalizeBook(
        parseJson(getLocal(BOOK_BACKUP_KEY))
      );

  var bookContext =
    sessionBookContext ||
    backupBookContext;

  var legacyContext = normalizeLegacy(
    parseJson(getSession(LEGACY_RAW_KEY))
  );

  var context =
    bookContext ||
    legacyContext ||
    queryContext();

  if (!context) return;

  var fingerprint = returnFingerprint(context);
  var alreadyPrepared =
    getSession(PREPARED_KEY) === fingerprint;

  setSession(DISPLAY_KEY, JSON.stringify(context));

  /*
   * Rehydrate the same-tab book context when the checkout return
   * arrived in a different browsing context.
   */
  if (backupBookContext) {
    setSession(
      BOOK_RAW_KEY,
      JSON.stringify({
        version:1,
        sku:BOOK_SKU,
        productName:BOOK_NAME,
        source:context.source,
        createdAt:context.ts,
        expiresAt:context.ts + MAX_AGE_MS
      })
    );
    removeLocal(BOOK_BACKUP_KEY);
  }

  /*
   * Bridge the account-first book checkout context into the
   * legacy purchase context expected by the current GTM pipeline.
   * This prepares data only. It does not fire purchase.
   */
  if (bookContext) {
    setSession(
      LEGACY_RAW_KEY,
      JSON.stringify(legacyPayload(context))
    );
  }

  if (!alreadyPrepared) {
    setSession(PREPARED_KEY, fingerprint);
    removeSession(FIRED_KEY);

    if (!legacyContext && !bookContext) {
      setSession(
        LEGACY_RAW_KEY,
        JSON.stringify(legacyPayload(context))
      );
    }
  }
})();
</script>

      if (
        url.protocol !== "http:" &&
        url.protocol !== "https:"
      ) {
        return "";
      }

      return url.pathname + url.search + url.hash;
    } catch (_) {
      return "";
    }
  }

  function setText(root, selector, value){
    var element = root.querySelector(selector);
    if (element) element.textContent = value;
  }

  function setProductMeta(root, value){
    var element = root.querySelector(
      "[data-nf-purchase-meta]"
    );

    if (!element) return;

    element.textContent = value || "";
    element.hidden = !value;
  }

  function setAction(root, index, config){
    var action = root.querySelector(
      '[data-nf-purchase-action-slot="' +
      String(index) +
      '"]'
    );

    if (!action) return;

    if (!config) {
      action.hidden = true;
      action.removeAttribute("href");
      action.removeAttribute("data-dest");
      action.removeAttribute("data-nf-purchase-action");
      return;
    }

    action.hidden = false;
    action.textContent = config.label;
    action.setAttribute("href", config.href);
    action.setAttribute("data-dest", config.href);
    action.setAttribute(
      "data-nf-purchase-action",
      config.action
    );

    action.classList.toggle(
      "nf-purchase-action--primary",
      !!config.primary
    );
  }

  function setActions(root, actions){
    [1,2,3].forEach(function(index){
      setAction(root, index, actions[index - 1] || null);
    });
  }

  function unwrapMember(result){
    var data = result;
    if (
      data &&
      Object.prototype.hasOwnProperty.call(data, "data")
    ) {
      data = data.data;
    }
    if (data && data.member) data = data.member;
    return data && typeof data === "object"
      ? data
      : null;
  }

  function connectionInactive(connection){
    if (!connection || typeof connection !== "object") {
      return false;
    }

    if (connection.active === false) return true;

    var status = safeText(
      connection.status ||
      (
        connection.payment &&
        connection.payment.status
      ) ||
      "",
      40
    ).toLowerCase();

    return [
      "canceled",
      "cancelled",
      "expired",
      "inactive"
    ].indexOf(status) >= 0;
  }

  function containsExactId(value, targets, depth){
    if (depth > 6 || value == null) return false;

    if (typeof value === "string") {
      return targets.indexOf(value) >= 0;
    }

    if (Array.isArray(value)) {
      for (var i = 0; i < value.length; i += 1) {
        if (
          containsExactId(
            value[i],
            targets,
            depth + 1
          )
        ) {
          return true;
        }
      }
      return false;
    }

    if (typeof value === "object") {
      var keys = Object.keys(value);

      for (var j = 0; j < keys.length; j += 1) {
        var child = value[keys[j]];

        if (
          typeof child === "string" &&
          targets.indexOf(child) >= 0
        ) {
          return true;
        }

        if (
          child &&
          typeof child === "object" &&
          containsExactId(
            child,
            targets,
            depth + 1
          )
        ) {
          return true;
        }
      }
    }

    return false;
  }

  function memberHasTargets(member, targets){
    if (!member || typeof member !== "object") {
      return false;
    }

    var collections = [];

    [
      member.planConnections,
      member.plans,
      member.plan_connections,
      member.memberships
    ].forEach(function(candidate){
      if (Array.isArray(candidate)) {
        collections.push(candidate);
      } else if (
        candidate &&
        Array.isArray(candidate.data)
      ) {
        collections.push(candidate.data);
      }
    });

    for (var i = 0; i < collections.length; i += 1) {
      for (var j = 0; j < collections[i].length; j += 1) {
        var connection = collections[i][j];

        if (connectionInactive(connection)) continue;

        if (
          containsExactId(
            connection,
            targets,
            0
          )
        ) {
          return true;
        }
      }
    }

    return false;
  }

  async function waitForMemberstack(timeoutMs){
    var end = Date.now() + (timeoutMs || 6000);

    while (Date.now() < end) {
      if (
        window.$memberstackDom &&
        typeof window.$memberstackDom.getCurrentMember ===
          "function"
      ) {
        return window.$memberstackDom;
      }
      await sleep(120);
    }

    return null;
  }

  async function recoverAccountAccess(){
    var ms = await waitForMemberstack(6000);

    if (!ms) {
      return {
        state:"unavailable",
        member:null,
        book:false,
        automaticPlan:false
      };
    }

    try {
      var result = await ms.getCurrentMember();
      var member = unwrapMember(result);

      if (!member) {
        return {
          state:"logged-out",
          member:null,
          book:false,
          automaticPlan:false
        };
      }

      var book = memberHasTargets(
        member,
        [BOOK_PLAN_ID, BOOK_PRICE_ID]
      );

      var automaticPlan = memberHasTargets(
        member,
        [
          AUTOMATIC_PLAN_PLAN_ID,
          AUTOMATIC_PLAN_PRICE_ID
        ]
      );

      return {
        state:"logged-in",
        member:member,
        book:book,
        automaticPlan:automaticPlan
      };
    } catch (_) {
      return {
        state:"unavailable",
        member:null,
        book:false,
        automaticPlan:false
      };
    }
  }

  function renderFromContext(root, context){
    var isBook = !!(
      context &&
      context.sku === BOOK_SKU
    );

    var isAutomaticPlan = !!(
      context &&
      AUTOMATIC_PLAN_SKUS[context.sku]
    );

    if (isBook) {
      renderBook(root, "purchase_context");
    } else if (isAutomaticPlan) {
      renderAutomaticPlan(
        root,
        "purchase_context"
      );
    } else if (context) {
      renderGeneric(root, context);
    }

    return {
      isBook:isBook,
      isAutomaticPlan:isAutomaticPlan
    };
  }

  onReady(async function(){
    var root = document.querySelector(
      "[data-nf-purchase-thankyou]"
    );

    if (!root) return;

    var context = parseContext();
    var contextState = {
      isBook:false,
      isAutomaticPlan:false
    };
    var recovery = null;
    var resolvedType = "unconfirmed";
    var verifiedBy = "none";

    if (context) {
      contextState = renderFromContext(
        root,
        context
      );
      resolvedType = contextState.isBook
        ? "book"
        : contextState.isAutomaticPlan
          ? "automatic_plan"
          : "purchase";
      verifiedBy = "purchase_context";
    } else {
      renderChecking(root);
      recovery = await recoverAccountAccess();

      if (
        recovery.state === "logged-in" &&
        recovery.book &&
        !recovery.automaticPlan
      ) {
        renderBook(root, "account_entitlement");
        resolvedType = "book";
        verifiedBy = "account_entitlement";
      } else if (
        recovery.state === "logged-in" &&
        recovery.automaticPlan &&
        !recovery.book
      ) {
        renderAutomaticPlan(
          root,
          "account_entitlement"
        );
        resolvedType = "automatic_plan";
        verifiedBy = "account_entitlement";
      } else if (
        recovery.state === "logged-in"
      ) {
        renderAccountAccess(root);
        resolvedType = "account_access";
        verifiedBy = "account_session";
      } else {
        renderUnconfirmed(root);
        resolvedType = "unconfirmed";
        verifiedBy = recovery
          ? recovery.state
          : "none";
      }
    }

    trackOnce(
      "view.purchase_thankyou",
      "nf_purchase_thankyou_view",
      {
        nf_purchase_context:!!context,
        nf_sku:context ? context.sku : null,
        nf_context_type:resolvedType,
        nf_automatic_plan:
          resolvedType === "automatic_plan",
        nf_book:resolvedType === "book",
        nf_verified_by:verifiedBy,
        nf_source_page:"purchase_thankyou",
        nf_path:
          window.location.pathname ||
          "/thank-you"
      }
    );

    document.addEventListener(
      "click",
      function(event){
        var target = event.target;
        var action =
          target &&
          target.closest
            ? target.closest(
                "[data-nf-purchase-action]"
              )
            : null;

        if (!action) return;

        track(
          "nf_purchase_thankyou_action_click",
          {
            nf_action:
              action.getAttribute(
                "data-nf-purchase-action"
              ) || "",
            nf_dest:safeInternalDest(
              action.getAttribute("href") ||
              action.getAttribute("data-dest") ||
              ""
            ),
            nf_purchase_context:!!context,
            nf_sku:context ? context.sku : null,
            nf_context_type:resolvedType,
            nf_verified_by:verifiedBy,
            nf_source_page:
              "purchase_thankyou"
          }
        );
      },
      { passive:true }
    );
  });
})();
