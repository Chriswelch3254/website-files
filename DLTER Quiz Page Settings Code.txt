function scaleScores() {
  var sums = {}, counts = {};
  SCORE_ORDER.forEach(function (scale) { sums[scale] = 0; counts[scale] = 0; });

  ITEMS.forEach(function (item) {
    var raw = Number(answers[item.id]);
    if (!(raw >= 1 && raw <= 6)) throw new Error("Every statement must be answered.");
    var adjusted = item.key === -1 ? 7 - raw : raw;
    sums[item.scale] += adjusted;
    counts[item.scale] += 1;
  });

  var scores = {};
  SCORE_ORDER.forEach(function (scale) {
    var mean = counts[scale] ? sums[scale] / counts[scale] : 3.5;
    scores[scale] = Math.round(clamp(((mean - 1) / 5) * 100, 0, 100) * 10) / 10;
  });
  return scores;
}

function archetypeDistance(scores, archetype) {
  var sum = 0, weightTotal = 0;
  for (var i = 0; i < SCORE_ORDER.length; i++) {
    var scale = SCORE_ORDER[i];
    var weight = SCORE_WEIGHTS[scale] || 1;
    var difference = Number(scores[scale]) - Number(archetype.profile[i]);
    sum += weight * difference * difference;
    weightTotal += weight;
  }
  return Math.sqrt(sum / weightTotal);
}

function classifyConvergence(firstDistance, secondDistance, scores) {
  var margin = secondDistance - firstDistance;
  var values = SCORE_ORDER.slice(0,6).map(function (scale) { return scores[scale]; });
  var max = Math.max.apply(Math, values);
  var min = Math.min.apply(Math, values);
  var differentiation = max - min;
  var label = "broad";
  if (firstDistance <= 23 && margin >= 5) label = "clear";
  else if (margin >= 2.75) label = "close";
  return {
    label: label,
    margin: Math.round(margin * 10) / 10,
    best_distance: Math.round(firstDistance * 10) / 10,
    differentiation: Math.round(differentiation * 10) / 10
  };
}

function domainSummary(scores, left, right) {
  var a = scores[left], b = scores[right], diff = Math.abs(a - b);
  var pattern = "balanced";
  if (a >= 65 && b >= 65) pattern = "dual";
  else if (diff >= 14) pattern = a > b ? "left" : "right";
  else if (a < 45 && b < 45) pattern = "low";
  return {left:a, right:b, difference:Math.round(diff * 10) / 10, pattern:pattern};
}

function responseQuality() {
  var values = ITEMS.map(function (item) { return Number(answers[item.id]); });
  var counts = {};
  values.forEach(function (v) { counts[v] = (counts[v] || 0) + 1; });
  var maxRun = 1, run = 1;
  for (var i = 1; i < values.length; i++) {
    if (values[i] === values[i - 1]) { run++; maxRun = Math.max(maxRun, run); }
    else run = 1;
  }
  var mostUsed = Math.max.apply(Math, Object.keys(counts).map(function (k) { return counts[k]; }));
  var mean = values.reduce(function (a,b) { return a+b; }, 0) / values.length;
  var variance = values.reduce(function (a,b) { var d=b-mean; return a+d*d; }, 0) / values.length;
  var status = "interpretable";
  if (mostUsed / values.length >= 0.75 || maxRun >= 18 || Math.sqrt(variance) < 0.55) status = "low_differentiation";
  return {
    status: status,
    longest_same_response_run: maxRun,
    most_used_response_count: mostUsed,
    response_sd: Math.round(Math.sqrt(variance) * 100) / 100
  };
}

function buildResultPacket() {
  var scores = scaleScores();
  var ranked = ARCHETYPES.map(function (archetype) {
    return {
      slug: archetype.slug,
      name: archetype.name,
      code: archetype.code,
      signature: archetype.signature,
      distance: archetypeDistance(scores, archetype)
    };
  }).sort(function (a,b) { return a.distance - b.distance; });

  var primary = ranked[0], secondary = ranked[1], tertiary = ranked[2];
  var convergence = classifyConvergence(primary.distance, secondary.distance, scores);
  var now = Date.now();

  return {
    schema: RESULT_SCHEMA,
    assessment_version: ASSESSMENT_VERSION,
    scoring_model_version: SCORING_VERSION,
    completed_at: new Date(now).toISOString(),
    completed_at_ms: now,
    expires_at_ms: now + RESULT_TTL_MS,
    duration_ms: Math.max(0, now - startedAt),
    type_slug: primary.slug,
    type_name: primary.name,
    type_code: primary.code,
    model_signature: primary.signature,
    primary: primary,
    resonances: [secondary, tertiary],
    scores: scores,
    domains: {
      sensemaking: domainSummary(scores, "RS", "EC"),
      stability: domainSummary(scores, "SA", "AR"),
      identity: domainSummary(scores, "CP", "TR")
    },
    convergence: convergence,
    response_quality: responseQuality()
  };
}

function persistUnsavedAttempt(record) {
  if (!record) return false;
  var serialized = "";
  try { serialized = JSON.stringify(record); } catch (e) { return false; }
  if (!serialized || utf8Bytes(serialized) > 24576) return false;
  var stored = safeStorageSet(localStorage, UNSAVED_KEY, serialized);
  safeStorageSet(sessionStorage, UNSAVED_KEY, serialized);
  return stored;
}

function clearUnsavedAttempt() {
  safeStorageRemove(localStorage, UNSAVED_KEY);
  safeStorageRemove(sessionStorage, UNSAVED_KEY);
}

function validUnsavedAttempt(record) {
  if (!isPlainObject(record) || record.schema !== UNSAVED_SCHEMA) return false;
  if (record.instrumentVersion !== INSTRUMENT_VERSION) return false;
  if (typeof record.clientAttemptId !== "string" || !/^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(record.clientAttemptId)) return false;
  if (!isFinite(Number(record.createdAt)) || !isFinite(Number(record.expiresAt)) || Date.now() > Number(record.expiresAt)) return false;
  if (!isIso(record.clientStartedAt) || !isIso(record.clientCompletedAt)) return false;
  if (!Array.isArray(record.responses) || record.responses.length !== ITEMS.length) return false;
  for (var i = 0; i < ITEMS.length; i++) {
    var row = record.responses[i];
    var expectedId = "rt_" + String(ITEMS[i].id).padStart(3, "0");
    if (!isPlainObject(row) || row.itemId !== expectedId || Number(row.position) !== i + 1) return false;
    if (!(Number(row.value) >= 1 && Number(row.value) <= 6)) return false;
    if (row.answeredAt != null && !isIso(row.answeredAt)) return false;
  }
  if (!isPlainObject(record.localResult) || record.localResult.schema !== RESULT_SCHEMA) return false;
  if (!ARCHETYPES.some(function (a) { return a.slug === record.localResult.type_slug; })) return false;
  return true;
}

function loadUnsavedAttempt() {
  var raw = safeStorageGet(sessionStorage, UNSAVED_KEY) || safeStorageGet(localStorage, UNSAVED_KEY);
  var record = safeJSONParse(raw);
  if (!validUnsavedAttempt(record)) {
    if (raw) clearUnsavedAttempt();
    return null;
  }
  return record;
}

function pruneUnsavedAttempt() {
  loadUnsavedAttempt();
}

      position: index + 1
    };
    if (isIso(answerTimes[item.id])) row.answeredAt = answerTimes[item.id];
    return row;
  });
}

function validServerResult(data) {
  if (!isPlainObject(data) || data.schema !== RESULT_SCHEMA || data.saved !== true) return false;
  if (typeof data.attemptId !== "string" || !/^[0-9a-f-]{36}$/i.test(data.attemptId)) return false;
  if (!isIso(data.completedAt) || !isPlainObject(data.versions) || data.versions.instrument !== INSTRUMENT_VERSION) return false;
  if (!isPlainObject(data.primary) || !ARCHETYPES.some(function (a) { return a.slug === data.primary.slug; })) return false;
  if (!Array.isArray(data.resonances) || data.resonances.length !== 2 || !isPlainObject(data.scores)) return false;
  for (var i = 0; i < SCORE_ORDER.length; i++) {
    var value = Number(data.scores[SCORE_ORDER[i]]);
    if (!isFinite(value) || value < 0 || value > 100) return false;
  }
  return true;
}

function legacyMatch(match) {
  return {
    slug: match.slug,
    name: match.name,
    code: match.legacyCode,
    signature: match.profileCode,
    distance: Number(match.distance)
  };
}

function legacyDomain(domain) {
  return {
    left: Number(domain.leftScore),
    right: Number(domain.rightScore),
    difference: Number(domain.difference),
    pattern: domain.pattern
  };
}

function adaptServerResult(data, localPacket, unsaved) {
  var completedMs = Date.parse(data.completedAt);
  var result = Object.assign({}, localPacket, {
    schema: RESULT_SCHEMA,
    assessment_version: ASSESSMENT_VERSION,
    instrument_version: data.versions.instrument,
    scoring_model_version: data.versions.scoring,
    archetype_version: data.versions.archetypes,
    completed_at: data.completedAt,
    completed_at_ms: completedMs,
    expires_at_ms: completedMs + RESULT_TTL_MS,
    client_attempt_id: unsaved.clientAttemptId,
    attempt_id: data.attemptId,
    saved: true,
    save_status: "saved",
    type_slug: data.primary.slug,
    type_name: data.primary.name,
    type_code: data.primary.legacyCode,
    model_signature: data.primary.profileCode,
    primary: legacyMatch(data.primary),
    resonances: [legacyMatch(data.resonances[0]), legacyMatch(data.resonances[1])],
    scores: data.scores,
    domains: {
      sensemaking: legacyDomain(data.domains.sensemaking),
      stability: legacyDomain(data.domains.stability),
      identity: legacyDomain(data.domains.identity)
    },
    convergence: {
      label: data.convergence.label,
      margin: Number(data.convergence.margin),
      best_distance: Number(data.convergence.firstDistance),
      differentiation: Number(data.differentiation)
    },
    response_quality: {
      status: data.quality.status,
      longest_same_response_run: Number(data.quality.longestRun),
      most_used_response_count: Math.round(Number(data.quality.categoryShare) * ITEMS.length),
      response_sd: Number(data.quality.rawStandardDeviation)
    }
  });
  return result;
}

async function saveAttempt(unsaved, localPacket) {
  if (!validUnsavedAttempt(unsaved)) return {kind:"invalid"};
  var token = await sessionToken();
  if (!token) {
    track("nf_reality_types_save_result", { result: "guest_local_only" });
    return {kind:"guest"};
  }

  var controller = typeof AbortController === "function" ? new AbortController() : null;
  var timeout = setTimeout(function () { if (controller) controller.abort(); }, SAVE_TIMEOUT_MS);
  try {
    var response = await fetch(API_URL, {
      method: "POST",
      headers: {
        Authorization: "Bearer " + token,
        "Content-Type": "application/json",
        Accept: "application/json"
      },
      credentials: "omit",
      cache: "no-store",
      signal: controller ? controller.signal : undefined,
      body: JSON.stringify({
        instrumentVersion: unsaved.instrumentVersion,
        clientAttemptId: unsaved.clientAttemptId,
        sourceSurface: unsaved.sourceSurface,
        clientStartedAt: unsaved.clientStartedAt,
        clientCompletedAt: unsaved.clientCompletedAt,
        responses: unsaved.responses
      })
    });
    var raw = await response.text();
    if (utf8Bytes(raw) > MAX_API_RESPONSE_BYTES) throw new Error("save_response_too_large");
    var data = safeJSONParse(raw);
    if ((response.status === 200 || response.status === 201) && validServerResult(data)) {
      track("nf_reality_types_save_result", { result: "saved" });
      return {kind:"saved", packet:adaptServerResult(data, localPacket, unsaved)};
    }
    var reason = data && typeof data.error === "string" ? data.error : "";
    if (response.status === 401) {
      track("nf_reality_types_save_result", { result: "auth_required" });
      return {kind:"auth", reason:reason};
    }
    if (response.status === 409) {
      track("nf_reality_types_save_result", { result: "idempotency_conflict" });
      return {kind:"conflict", reason:reason};
    }
    if (response.status === 503) {
      track("nf_reality_types_save_result", { result: "temporarily_unavailable" });
      return {kind:"unavailable", reason:reason};
    }
    track("nf_reality_types_save_result", { result: "not_saved" });
    return {kind:"invalid", reason:reason};
  } catch (e) {
    track("nf_reality_types_save_result", { result: "network_retryable" });
    return {kind:"retry"};
  } finally {
    clearTimeout(timeout);
  }
}

function handleSaveOutcome(outcome) {
  var kind = outcome && outcome.kind ? outcome.kind : "retry";
  if (kind === "saved" && outcome.packet) {
    currentCompletionPacket = outcome.packet;
    persistResult(currentCompletionPacket);
    clearUnsavedAttempt();
    currentUnsavedAttempt = null;
    showSavePanel("saved");
    return;
  }
  showSavePanel(kind, outcome && outcome.reason ? outcome.reason : "");
}

async function retryCurrentSave() {
  if (saveInFlight || !currentUnsavedAttempt || !currentCompletionPacket) return;
  saveInFlight = true;
  showSaveBusy("Saving your Reality Map");
  try {
    handleSaveOutcome(await saveAttempt(currentUnsavedAttempt, currentCompletionPacket));
  } finally {
    saveInFlight = false;
    if (saveCard) saveCard.classList.remove("is-busy");
  }
}

function stopAuthRecoveryPolling() {
  if (authRecoveryTimer) {
    clearInterval(authRecoveryTimer);
    authRecoveryTimer = null;
  }
}

function beginAuthRecoveryPolling() {
  authRecoveryRequested = true;
  stopAuthRecoveryPolling();
  var attempts = 0;
  authRecoveryTimer = setInterval(async function () {
    attempts += 1;
    if (!authRecoveryRequested || attempts > 80) {
      stopAuthRecoveryPolling();
      return;
    }
    var token = await sessionToken();
    if (token) {
      authRecoveryRequested = false;
      stopAuthRecoveryPolling();
      retryCurrentSave();
    }
  }, 750);
}

function requestAuthAndSave(mode) {
  authRecoveryRequested = true;
  var opened = openMemberstackModal(mode);
  beginAuthRecoveryPolling();
  return opened;
}

async function completeAssessment() {
  if (saveInFlight) return;
  if (answeredCount() !== ITEMS.length) {
    showReview();
    return;
  }

  assessmentCompleted = true;
  if (loadingEl) loadingEl.hidden = false;
  if (form) form.hidden = true;
  if (saveCard) saveCard.hidden = true;
  if (errorEl) errorEl.textContent = "";

  var packet;
  var unsaved = null;
  try {
    packet = buildResultPacket();
    packet.instrument_version = INSTRUMENT_VERSION;
    packet.saved = false;
    packet.save_status = "local_only";
    unsaved = buildUnsavedAttempt(packet);
    if (!unsaved) {
      throw new Error("This browser could not create a secure attempt identifier. Please try another browser before continuing.");
    }
    packet.client_attempt_id = unsaved.clientAttemptId;
    if (!persistUnsavedAttempt(unsaved)) {
      throw new Error("This browser could not create a safe local recovery copy. Please free storage space or try another browser before continuing.");
    }
    currentCompletionPacket = packet;
    currentUnsavedAttempt = unsaved;
    persistResult(packet);
    clearProgress();
  } catch (e) {
    assessmentCompleted = false;
    if (loadingEl) loadingEl.hidden = true;
    if (form) form.hidden = false;
    if (errorEl) errorEl.textContent = e && e.message ? e.message : "The result could not be calculated. Please try again.";
    return;
  }

  track("nf_reality_types_assessment_complete", {
    type_slug: packet.type_slug,
    type_code: packet.type_code,
    convergence: packet.convergence.label,
    response_quality: packet.response_quality.status
  });

  saveInFlight = true;
  try {
    handleSaveOutcome(await saveAttempt(unsaved, packet));
  } finally {
    saveInFlight = false;
  }
}

}

function onKeydown(event) {
  if (!root || !root.contains(document.activeElement)) return;
  if (event.altKey || event.ctrlKey || event.metaKey) return;
  var key = event.key || "";
  if (/^[1-6]$/.test(key) && stage && stage.querySelector(".rt-question-card")) {
    var radio = stage.querySelector('input[value="' + key + '"]');
    if (radio) {
      event.preventDefault();
      radio.checked = true;
      radio.dispatchEvent(new Event("change", {bubbles:true}));
    }
  }
}

  errorEl = document.getElementById("rt-error");
  loadingEl = document.getElementById("rt-loading");
  saveCard = document.getElementById("rt-save-card");
  saveEyebrow = document.getElementById("rt-save-eyebrow");
  saveTitle = document.getElementById("rt-save-title");
  saveCopy = document.getElementById("rt-save-copy");
  saveDetail = document.getElementById("rt-save-detail");
  saveRetryBtn = document.getElementById("rt-save-retry");
  saveSignupBtn = document.getElementById("rt-save-signup");
  saveLoginBtn = document.getElementById("rt-save-login");
  saveContinueBtn = document.getElementById("rt-save-continue");

  if (!form || !stage || !backBtn || !nextBtn) return;

  if (progressTrack) {
    progressTrack.setAttribute("aria-valuenow", "0");
    progressTrack.setAttribute("aria-valuetext", "0 of " + ITEMS.length + " statements answered");
  }

  pruneUnsavedAttempt();

  var saved = loadProgress();
  var pending = loadUnsavedAttempt();
  if (saved) {
    answers = saved.answers;
    answerTimes = saved.answer_times || {};
    currentIndex = saved.current_index;
    startedAt = saved.started_at;
    if (resumeBanner) resumeBanner.hidden = false;
    if (resumeCopy) resumeCopy.textContent = answeredCount() + " of " + ITEMS.length + " statements are already answered in this tab.";
  } else if (pending) {
    assessmentCompleted = true;
    currentUnsavedAttempt = pending;
    currentCompletionPacket = pending.localResult;
    persistResult(currentCompletionPacket);
    if (form) form.hidden = true;
    showSavePanel("recovery");
  }

  backBtn.addEventListener("click", function () {
    if (currentIndex > 0) goTo(currentIndex - 1);
  });

  nextBtn.addEventListener("click", function () {
    var item = itemAt(currentIndex);
    if (!item || !answers[item.id]) {
      if (errorEl) errorEl.textContent = "Choose the response that fits you best before continuing.";
      return;
    }
    if (errorEl) errorEl.textContent = "";
    if (currentIndex === ITEMS.length - 1) showReview();
    else goTo(currentIndex + 1);
  });

  if (resumeContinueBtn) resumeContinueBtn.addEventListener("click", function () {
    if (resumeBanner) resumeBanner.hidden = true;
    goTo(currentIndex);
  });

  if (startOverBtn) startOverBtn.addEventListener("click", startOver);
  if (saveRetryBtn) saveRetryBtn.addEventListener("click", retryCurrentSave);
  if (saveSignupBtn) saveSignupBtn.addEventListener("click", function (event) {
    if (requestAuthAndSave("signup")) {
      event.preventDefault();
      event.stopPropagation();
    }
  });
  if (saveLoginBtn) saveLoginBtn.addEventListener("click", function (event) {
    if (requestAuthAndSave("login")) {
      event.preventDefault();
      event.stopPropagation();
    }
  });
  if (saveContinueBtn) saveContinueBtn.addEventListener("click", continueToRealityMap);

  function retryAfterExplicitAuth() {
    if (!authRecoveryRequested) return;
    beginAuthRecoveryPolling();
  }
  document.addEventListener("memberstack:loggedIn", retryAfterExplicitAuth);
  window.addEventListener("memberstack:loggedIn", retryAfterExplicitAuth);
  window.addEventListener("focus", retryAfterExplicitAuth);
  document.addEventListener("visibilitychange", function () {
    if (document.visibilityState === "visible") retryAfterExplicitAuth();
  });

  document.addEventListener("keydown", onKeydown);
  window.addEventListener("pagehide", function () {
    stopAuthRecoveryPolling();
    clearTimeout(redirectTimer);
    if (!assessmentCompleted && !saveInFlight && form && !form.hidden && answeredCount() < ITEMS.length) saveProgress();
  });

  if (!assessmentCompleted) renderQuestion();
  track("nf_reality_types_assessment_view", {});
}

if (document.readyState === "loading") {
  document.addEventListener("DOMContentLoaded", init, {once:true});
} else {
  init();
}
})();
