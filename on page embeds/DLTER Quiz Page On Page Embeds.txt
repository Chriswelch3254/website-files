(function () {
  "use strict";

  if (window.__NF_RT_PRESENTATION_322__) return;
  window.__NF_RT_PRESENTATION_322__ = true;

  var TYPE_RENAMES = {
    "The Structured Empathic Lens": "The Structured Insight Lens",
    "Structured Empathic Lens": "Structured Insight Lens",
    "The Resonant Fieldholder": "The Resonant Integrator",
    "Resonant Fieldholder": "Resonant Integrator"
  };

  function canonicalString(value) {
    return Object.prototype.hasOwnProperty.call(TYPE_RENAMES, value) ? TYPE_RENAMES[value] : value;
  }

  function canonicalize(value) {
    if (typeof value === "string") return canonicalString(value);
    if (Array.isArray(value)) return value.map(canonicalize);
    if (value && typeof value === "object") {
      Object.keys(value).forEach(function (key) {
        value[key] = canonicalize(value[key]);
      });
    }
    return value;
  }

  function canonicalizeStoredValue(storage, key) {
    try {
      var raw = storage.getItem(key);
      if (!raw) return;
      var parsed = JSON.parse(raw);
      var updated = JSON.stringify(canonicalize(parsed));
      if (updated !== raw) storage.setItem(key, updated);
    } catch (_) {}
  }

  function canonicalizeLegacyLocalResults() {
    var keys = ["nf.realityTypes.lastResult.v2", "nf.realityTypes.unsavedAttempt.v1"];
    [window.localStorage, window.sessionStorage].forEach(function (storage) {
      if (!storage) return;
      keys.forEach(function (key) {
        canonicalizeStoredValue(storage, key);
      });
    });

    try {
      if (window.__DLTER_LAST_RESULT__) canonicalize(window.__DLTER_LAST_RESULT__);
    } catch (_) {}
  }

  function placeLibrary() {
    var section = document.getElementById("reality-types-library");
    var assessment = document.querySelector("[data-rt-assessment]");
    if (!section || !assessment || section.__rtPlaced) return;
    assessment.insertAdjacentElement("afterend", section);
    section.__rtPlaced = true;
  }

  function unlockLibrary() {
    var section = document.getElementById("reality-types-library");
    var details = section && section.querySelector("[data-rt-type-library-details]");
    if (!section || !details || section.__rtUnlocked) return;

    section.__rtUnlocked = true;
    section.classList.remove("is-locked");

    var summary = details.querySelector("summary");
    if (summary) summary.removeAttribute("aria-disabled");

    var title = section.querySelector("#rt-type-library-title");
    var subtitle = section.querySelector("[data-rt-type-library-subtitle]");
    var toggle = section.querySelector(".rt-type-library-toggle");

    if (title) title.textContent = "Explore the 12 Reality Types";
    if (subtitle) subtitle.textContent = "See the reference patterns your Reality Map may resemble.";
    if (toggle) toggle.textContent = "";
  }

  function bindLibraryLock() {
    var section = document.getElementById("reality-types-library");
    var details = section && section.querySelector("[data-rt-type-library-details]");
    var summary = details && details.querySelector("summary");
    if (!section || !details || !summary || summary.__rtLockBound) return;

    summary.__rtLockBound = true;
    summary.setAttribute("aria-disabled", "true");

    summary.addEventListener("click", function (event) {
      if (!section.__rtUnlocked) {
        event.preventDefault();
        details.open = false;
      }
    });
  }

  function optionGroup(title, subtitle, options, className) {
    var group = document.createElement("div");
    group.className = "rt-response-group " + className;
    group.setAttribute("role", "group");
    group.setAttribute("aria-label", title + " responses");

    var head = document.createElement("div");
    head.className = "rt-response-group-head";

    var strong = document.createElement("strong");
    strong.textContent = title;

    var span = document.createElement("span");
    span.textContent = subtitle;

    head.appendChild(strong);
    head.appendChild(span);

    var scale = document.createElement("div");
    scale.className = "rt-scale";
    options.forEach(function (option) {
      scale.appendChild(option);
    });

    group.appendChild(head);
    group.appendChild(scale);
    return group;
  }

      review.querySelectorAll(".rt-review-stat span").forEach(function (node) {
        if (node.textContent.trim() === "archetype comparisons") {
          node.textContent = "reference-pattern comparisons";
        }
      });

      var reviewParagraph = review.querySelector("p:not(.rt-question-number)");
      if (reviewParagraph && reviewParagraph.textContent.indexOf("similarities across all 12 Reality Types") !== -1) {
        reviewParagraph.textContent =
          "Your responses will be converted into six mode scores, three coordination scores, and comparisons with all 12 reference patterns. The result is an interpretive map, not a diagnosis or permanent label.";
      }
      return;
    }

    var card = stage.querySelector(".rt-question-card");
    if (!card || card.__rt322Enhanced) return;

    var oldScale = card.querySelector(".rt-scale");
    if (!oldScale) return;

    var options = Array.prototype.slice.call(oldScale.querySelectorAll(".rt-option"));
    if (options.length !== 6) return;

    card.__rt322Enhanced = true;

    var help = card.querySelector("#rt-scale-help");
    if (help) {
      help.textContent = "Choose the response that best reflects your usual pattern.";
    }

    if (!stage || stage.__rt322Observer) return;

    stage.__rt322Observer = true;
