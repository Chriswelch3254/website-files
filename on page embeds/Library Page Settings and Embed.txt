  function renderAction(action,item,statusNode){
    if(!action) return null;
    if(action.kind === "route"){
      var routeAttrs = {
        class:action.primary === false ? "nf-link" : "nf-btn nf-btn-primary",
        href:action.href || CONFIG.routes.support,
        target:action.target === "_blank" ? "_blank" : "_self",
        rel:action.target === "_blank" ? "noopener noreferrer" : ""
      };
      if(action.target === "_blank") routeAttrs["aria-label"] = (action.label || "Open") + ", opens in a new tab";
      routeAttrs["data-nf-action-key"] = action.actionKey || "";
      return el("a",routeAttrs,[action.label || "Open"]);
    }
    if(action.kind === "protected_static"){
      var protectedAttrs = {class:action.primary === false ? "nf-link" : "nf-btn nf-btn-primary",type:"button"};
      if(action.mode !== "download") protectedAttrs["aria-label"] = (action.label || "Open") + ", opens in NeuForm document tab";
      protectedAttrs["data-nf-action-key"] = action.actionKey || "";
      var button = el("button",protectedAttrs,[action.label || "Open"]);
      button.addEventListener("click",function(){openStaticPdf(item,action,button,statusNode);});
      return button;
    }
    if(action.kind === "custom" && typeof action.run === "function"){
      var custom = el("button",{class:action.primary === false ? "nf-link" : "nf-btn nf-btn-primary",type:"button","data-nf-action-key":action.actionKey || ""},[action.label || "Continue"]);
      custom.addEventListener("click",function(){rememberFocus(item,action);action.run({button:custom,statusNode:statusNode,item:item});});
      return custom;
    }
    return null;
  }

  function statusFlow(item){

    var current = item.progressCurrent >= 0 && item.progressCurrent < item.progressSteps.length ? item.progressCurrent : -1;
    var list = el("div",{class:"nf-status-flow",role:"list","aria-label":"Automatic Plan progress: " + item.progressLabel},item.progressSteps.map(function(label,index){
      var stateName = index < completed ? "complete" : index === current ? "current" : "upcoming";
      var attrs = {class:"nf-status-step",role:"listitem","data-state":stateName};
      if(stateName === "current") attrs["aria-current"] = "step";
      var detail = stateName === "complete" ? ", completed" : stateName === "current" ? ", current step" : ", not started";
      return el("span",attrs,[el("span",{},[label]),el("span",{class:"nf-sr-only"},[detail])]);

    if(current !== "all" && !available.has(current)){

      NF.state.filters.section = "all";

    state.ui.select.value = current;
