function lockNameText(root,def){

];

var value=String(node.nodeValue||"");

}});

nodes.forEach(function(textNode){
var value=String(textNode.nodeValue||""),frag=document.createDocumentFragment(),cursor=0,changed=false;

var best=null,bestIndex=-1;
for(var i=0;i<candidates.length;i++){var at=value.indexOf(candidates[i].from,cursor);if(at!==-1&&(bestIndex===-1||at<bestIndex)){best=candidates[i];bestIndex=at}}
