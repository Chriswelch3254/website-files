function cloudSaveStatus(cloud, storage, accessState) {
  cloud = cloud || {};
  storage = storage || {};
  var pending = cloud.pendingLocal === true || Number(cloud.queued) > 0 || cloud.localSavePending || cloud.localEditsPending;
  if (storage.state === "storage-warning" || storage.state === "storage-limited") return { state: "attention", title: "Needs attention", detail: "This browser could not safely keep your changes. Keep this tab open and download your work in Saving settings." };
  if (cloud.restoring) return { state: "saving", title: "Updating…", detail: "Opening your latest saved workspace." };
  if (cloud.workspaceReviewPending) return { state: "attention", title: "Needs attention", detail: "Another tab has newer changes. Your current work is retained for review." };
  if (cloud.recoveryPartial) return { state: "attention", title: "Needs attention", detail: "Some records need recovery. Open Saving settings to review them." };
  if (accessState && accessState.kind === "guest" && !cloud.localSavePending && !cloud.localEditsPending) return { state: "device", title: "Saved on this device", detail: "Sign in for automatic account saving. Existing guest records stay separate." };
  if (cloud.automaticPaused || cloud.saveState === "paused") return { state: "paused", title: "Account saving paused", detail: "Changes stay on this device. Resume in Saving settings to update your account." };
  if (cloud.saveState === "attention") return { state: "attention", title: "Needs attention", detail: "Your work is retained. Open Saving settings to finish syncing." };
  if (cloud.saveState === "offline") return { state: "offline", title: "Offline", detail: cloud.localSavePending ? "Finishing the device save. Keep this tab open." : "Changes saved on this device will sync automatically when you reconnect." };
  if (pending || cloud.saveState === "saving") return { state: "saving", title: "Saving…", detail: cloud.available && cloud.consent ? "Your changes are being saved to your account." : "Your changes are being saved on this device." };
  if (cloud.available && cloud.consent && cloud.currentAcknowledged === true) return { state: "saved", title: "Saved", detail: "Your latest changes are saved to your account and available on your other devices." };
  if (accessState && accessState.kind === "guest") return { state: "device", title: "Saved on this device", detail: "Sign in for automatic account saving. Existing guest records stay separate." };
  if (cloud.available && !cloud.consent) return { state: "checking", title: "Connecting…", detail: "Preparing automatic account saving. Your device keeps your current work." };
  return { state: "checking", title: "Connecting…", detail: "Checking your account and latest saved work." };
}

function dateLabel(value) {
  var date = new Date(value);
  return value && Number.isFinite(date.getTime()) ? date.toLocaleString(undefined, { year: "numeric", month: "short", day: "numeric", hour: "numeric", minute: "2-digit" }) : "Not recorded";
}

function trainingCopy(title, value, empty) {
  var fields = [["Lifts", "lifts"], ["Logged sets", "sets"], ["Sessions", "sessions"]];
  function count(key) {
    if (empty) return 0;
    return value && Number.isSafeInteger(value[key]) && value[key] >= 0 ? value[key] : "Not available";
  }
  return [
    "<article class=\"nfsl-v3-cloud-copy\"><h4>" + esc(title) + "</h4>",
    empty ? "<p>No training records saved yet.</p>" : "",
    "<dl>" + fields.map(function(field) { return "<div><dt>" + esc(field[0]) + "</dt><dd>" + esc(count(field[1])) + "</dd></div>"; }).join("") + "</dl>",
    "<p class=\"nfsl-v3-cloud-copy-updated\">Last saved: " + esc(dateLabel(value && value.updatedAt)) + "</p></article>"
  ].join("");
}

    return "<li><div><strong>Saved version " + esc(item.revision) + "</strong><span>" + esc(dateLabel(item.createdAt || item.updatedAt)) + "</span></div>" + cloudButton("history-export", "Download version " + item.revision, enabled && Number.isSafeInteger(item.revision) && item.revision > 0, " data-cloud-revision=\"" + attr(item.revision) + "\"") + "</li>";

No subscription changes here.</p><div class=\"nfsl-v3-actions\"><a class=\"nfsl-v3-btn nfsl-v3-btn--primary\" href=\"/support?topic=strength-lab\">Get both-Labs help</a>" + (accessState.kind === "guest" ? "" : "<button type=\"button\" class=\"nfsl-v3-btn\" data-action=\"manage-billing\">Manage billing</button>") + "</div>";
  } else if (recommendation.action === "manage") {
    var hasBundle = commercial.productAccess(accessState.kind).bundle;
    var text = bundle && !hasBundle ? "You already have a paid Lab subscription. Review your existing plan in billing or contact support before changing to both Labs. Your current subscription will not be changed here." : "Your account already has " + title + " access.";
    content = "<p>" + esc(text) + "</p><div class=\"nfsl-v3-actions\"><button type=\"button\" class=\"nfsl-v3-btn nfsl-v3-btn--primary\" data-action=\"manage-billing\">Manage billing</button><a class=\"nfsl-v3-btn\" href=\"/support?topic=strength-lab\">Get billing help</a></div>";
  } else {
    var guest = accessState.kind === "guest";
    content = "<div class=\"nfsl-v3-offer-grid\">" + offer(monthly, "Billed monthly.", guest, ui.offerInterval === "monthly", ui.pass6CheckoutBusy) + offer(annual, "Billed annually. " + commercial.deriveSavings(annual.id).savingsLabel + ".", guest, ui.offerInterval === "annual", ui.pass6CheckoutBusy) + "</div>";
  }
  return "<div class=\"nfsl-v3-pass6-offers\" data-nfsl-commercial-offer><p>" + esc(intro) + "</p>" + content + "</div>" + (ui.pass6CommercialText ? "<p role=\"status\" class=\"nfsl-v3-note\" data-nfsl-commercial-status>" + esc(ui.pass6CommercialText) + "</p>" : "") + "<p class=\"nfsl-v3-note\">Subscriptions renew until canceled. Checkout shows the final terms before you pay. Canceling Pro does not itself delete saved records.</p>";
}
