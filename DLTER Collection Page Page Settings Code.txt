    if (message) announce(message);
  }

  function beginSignup() {
    var client;
    var result;
    if (runtime.signupBusy || runtime.busyControl) return;
    if (runtime.memberStatus !== "known" || runtime.memberPresent) {
      revalidateMember();
      announce("Checking your NeuForm account before opening sign-in.");
      return;
    }
    client = memberstackWith("openModal");
    if (!client || typeof client.openModal !== "function") {
      startCapabilityPoll();
      reconcileControls();
      announce("Account sign-in is temporarily unavailable. Refresh the page or open Support if the problem continues.");
      return;
    }

    runtime.signupBusy = true;
    setCheckoutState("signup-opening");
    saveReturnTarget();
    reconcileControls();
    announce("Opening NeuForm account sign-in...");

    try {
      result = client.openModal("signup");
      if (result && typeof result.catch === "function") {result.catch(function () {
          restoreSignup("Account sign-in did not open. Please try again.");
        });
      }
    } catch (_) {
      restoreSignup("Account sign-in did not open. Please try again.");
      return;
    }

    runtime.signupTimer = window.setTimeout(function () {
      runtime.signupTimer = 0;
      runtime.signupBusy = false;
      setCheckoutState("idle");
      reconcileControls();
    }, SIGNUP_DEBOUNCE_MS);
  }

  function messageForBlockedState(state) {
    if (state === "member-checking") return "Checking your NeuForm account before enabling the next step.";
    if (state === "account-unavailable") return "Your NeuForm account could not be verified. Refresh the page or open Support. Your saved result has not changed.";
    if (state === "checkout-unavailable") return "Secure checkout is temporarily unavailable. Your saved result remains available in My Library.";

    if (state === "price-mismatch") return "Blueprint checkout is temporarily unavailable. Your saved result remains saved. Open Support for help.";
    if (state === "result-incomplete") return "Your complete Reality Map is still loading. Checkout remains unavailable until all nine scores and the closest match are ready.";
    if (state === "not-ready" || state === "not-saved-result") return "Blueprint access is still being checked. Please try again in a moment.";
    return "This Blueprint action is not available in the current page state.";
  }

  function handleCheckoutClick(event) {
    var target = event.target;
    var control = target && typeof target.closest === "function" ? target.closest(CHECKOUT_SELECTOR) : null;
    var state;
    if (!control) return;

    event.preventDefault();
    event.stopPropagation();
    if (typeof event.stopImmediatePropagation === "function") event.stopImmediatePropagation();

    state = controlState();
    reconcileControls();

    if (state === "signup-required") {
      beginSignup();
      return;
    }
    if (state === "eligible") {
      beginCheckout(control);
      return;
    }
    announce(messageForBlockedState(state));
  }

    runtime.memberSync += 1;

    document.dispatchEvent(new CustomEvent("rt:templateShellReady", { detail: { version: VERSION } }));

<script id="nf-rt-current-name-observer-v2">
(function(){
  "use strict";
  var MAP={
    "structured-empathic-lens":"The Structured Insight Lens",
    "resonant-fieldholder":"The Resonant Integrator"
  };
  function apply(){
    if(!document.body)return;
    if((document.body.getAttribute("data-rt-presentation-state")||"")==="historical-purchase")return;
    var slug=document.body.getAttribute("data-reality-type")||document.documentElement.getAttribute("data-wf-item-slug")||"";
    var name=MAP[slug];
    if(!name)return;
    document.querySelectorAll("#dlter-type-title,[data-rt-public-name],.dlter-type-name").forEach(function(el){
      if((el.textContent||"").trim()!==name)el.textContent=name;
    });
  }
  function boot(){
    apply();
    var title=document.getElementById("dlter-type-title");
    if(title && !title.__nfNameWatch){
      var watcher=new MutationObserver(function(){apply();});
      watcher.observe(title,{subtree:true,childList:true,characterData:true});
      title.__nfNameWatch=watcher;
    }
    var root=document.getElementById("rt-overview-view")||document.body;
    if(root && !root.__nfNameRootWatch){
      var rootWatch=new MutationObserver(function(records){
        var relevant=records.some(function(r){
          return r.type==="childList"||r.type==="characterData";
        });
        if(relevant)apply();
      });
      rootWatch.observe(root,{subtree:true,childList:true,characterData:true});
      root.__nfNameRootWatch=rootWatch;
    }
  }
  if(document.readyState==="loading")document.addEventListener("DOMContentLoaded",boot,{once:true});
  else boot();
  setTimeout(boot,500);
})();
</script>
