<script>
(function(){
  "use strict";

  var SOURCE_SELECTOR="[data-nf-rte-source-wrapper]";
  var READY_SELECTOR="[data-nf-blog-post]";

  function source(){
    return document.querySelector(SOURCE_SELECTOR);
  }

  function root(){
    return document.querySelector(READY_SELECTOR);
  }

  function hideSource(){
    var el=source();
    if(el) el.setAttribute("aria-hidden","true");
  }

  function ready(){
    var el=root();
    return !!(el&&el.getAttribute("data-nf-body-ready")==="true");
  }

  function restoreNormal(){
    document.documentElement.classList.remove("nf-article-failopen");
    hideSource();
  }

  hideSource();

  var observer=null;
  if("MutationObserver" in window){
    observer=new MutationObserver(function(){
      hideSource();
      if(ready()){
        restoreNormal();
        try{observer.disconnect();}catch(_){}
      }
    });

    try{
      observer.observe(document.documentElement,{
        childList:true,
        subtree:true,
        attributes:true,
        attributeFilter:["data-nf-body-ready"]
      });
    }catch(_){
      observer=null;
    }
  }

  window.setTimeout(function(){
    if(ready()){
      restoreNormal();
      return;
    }

    var el=source();
    if(!el) return;

    document.documentElement.classList.add("nf-article-failopen");
    el.removeAttribute("aria-hidden");
  },4500);
})();
</script>

    return false;
  }

  function trackOnce(key, name, payload){
    if (!canTrack()) return false;

    try {
      if (window.neuform && typeof window.neuform.trackOnce === "function") {
        return window.neuform.trackOnce(key, name, payload || {});
      }
    } catch(e) {}

    try {
      var storageKey = "nf.once." + key;
      if (window.sessionStorage && sessionStorage.getItem(storageKey)) return false;
      if (window.sessionStorage) sessionStorage.setItem(storageKey, "1");
    } catch(e) {}

    return track(name, payload || {});
  }

  function stripIds(node){
    if (!node) return;
    qsa("[id]", node).forEach(function(el){ el.removeAttribute("id"); });
  }

  function sanitizeRte(scope){
    if (!scope) return;

    qsa("script, style, link, meta, object, embed", scope).forEach(function(el){
      el.remove();
    });

    qsa("*", scope).forEach(function(el){
      Array.prototype.slice.call(el.attributes || []).forEach(function(attr){
        var name = attr.name.toLowerCase();
        var value = clean(attr.value);
        if (name.indexOf("on") === 0) el.removeAttribute(attr.name);
        if ((name === "href" || name === "src") && /^(javascript|data):/i.test(value)) {
          el.removeAttribute(attr.name);
        }
      });
    });

    qsa("iframe[src], img[src], video[src], audio[src], source[src]", scope).forEach(function(el){
      var src = validUrl(el.getAttribute("src"));
      if (!src) {
        el.remove();
        return;
      }
      el.setAttribute("src", src);
    });
  }

  function lazyMedia(scope){
    qsa("img", scope).forEach(function(img){
      if (!img.hasAttribute("loading")) img.setAttribute("loading", "lazy");
      if (!img.hasAttribute("decoding")) img.setAttribute("decoding", "async");
    });
    qsa("iframe", scope).forEach(function(frame){
      if (!frame.hasAttribute("loading")) frame.setAttribute("loading", "lazy");
    });
  }

    return !!qs("img, iframe, video, audio, picture, table, ul, ol, blockquote, pre", slot);
  }

  function headingId(text, used){
    var base = clean(text)
      .toLowerCase()
      .replace(/&/g, " and ")
      .replace(/[^a-z0-9]+/g, "-")
      .replace(/^-+|-+$/g, "")
      .slice(0, 72);

    if (!base) base = "section";

    var next = base;
    var index = 2;

    while (used[next]) {
      next = base + "-" + index;
      index += 1;
    }

    used[next] = true;
    return next;
  }

  function buildToc(){
    if (!root) return;

    var wrap = qs(TOC_WRAP_SELECTOR, root);
    var list = qs(TOC_LIST_SELECTOR, root);
    var body = qs(BODY_SLOT_SELECTOR, root);

    if (!wrap || !list || !body) return;

    var primaryHeadings = qsa("h2", body).filter(function(heading){
      return !!clean(heading.textContent);
    });
    var headings = primaryHeadings.length >= 2
      ? primaryHeadings
      : qsa("h2, h3", body).filter(function(heading){
          return !!clean(heading.textContent);
        });

    if (headings.length < 2) {
      hide(wrap);
      tocBuilt = true;
      return;
    }

    var used = {};
    list.innerHTML = "";

    headings.forEach(function(heading, index){
      var label = clean(heading.textContent);
      var id = clean(heading.id);

      if (!id || used[id]) {
        id = headingId(label, used);
        heading.id = id;
      } else {
        used[id] = true;
      }

      heading.setAttribute("tabindex", "-1");

      var item = document.createElement("li");
      item.className = heading.tagName.toLowerCase() === "h3"
        ? "nf-blog-post__toc-item nf-blog-post__toc-item--sub"
        : "nf-blog-post__toc-item";

      var link = document.createElement("a");
      link.href = "#" + id;
      link.textContent = label;
      link.setAttribute("data-toc-index", String(index + 1));

      link.addEventListener("click", function(event){
        event.preventDefault();

        var reduced = false;
        try {
          reduced = window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
        } catch(e) {}

        try {
          heading.scrollIntoView({
            behavior: reduced ? "auto" : "smooth",
            block: "start"
          });
        } catch(e) {
          window.location.hash = id;
        }

        try {
          history.replaceState(null, "", "#" + id);
        } catch(e) {}

        window.setTimeout(function(){
          try { heading.focus({ preventScroll: true }); } catch(e) {
            try { heading.focus(); } catch(_) {}
          }
        }, reduced ? 0 : 260);

        var payload = blogPayload();
        payload.toc_index = index + 1;
        addPayloadValue(payload, "toc_heading", safeAnalyticsText(label, 100));
        track("nf_blog_toc_click", payload);
      });

      item.appendChild(link);
      list.appendChild(item);
    });

    show(wrap);
    tocBuilt = true;
    updateTocCurrent();
  }

  function updateTocCurrent(){
    if (!root || root.getAttribute("data-nf-body-ready") !== "true") return;

    var body = qs(BODY_SLOT_SELECTOR, root);
    var list = qs(TOC_LIST_SELECTOR, root);
    if (!body || !list) return;

    var headings = qsa("h2[id]", body).filter(function(heading){
      return !!clean(heading.textContent);
    });
    var links = qsa('a[href^="#"]', list);
    if (!headings.length || !links.length) return;

    var navOffset = 96;
    try {
      var rawOffset = getComputedStyle(document.documentElement).getPropertyValue("--nf-site-nav-offset");
      var parsedOffset = parseFloat(rawOffset);
      if (Number.isFinite(parsedOffset)) navOffset = parsedOffset + 40;
    } catch(e) {}

    var current = headings[0];
    headings.forEach(function(heading){
      if (heading.getBoundingClientRect().top <= navOffset) current = heading;
    });

    var currentId = clean(current && current.id);
    links.forEach(function(link){
      var active = currentId && link.getAttribute("href") === "#" + currentId;
      if (active) link.setAttribute("aria-current", "location");
      else link.removeAttribute("aria-current");
    });
  }

  function updateReadingProgress(){
    if (!root || root.getAttribute("data-nf-body-ready") !== "true") return;

    var body = qs(BODY_SLOT_SELECTOR, root);
    var progress = qs(PROGRESS_SELECTOR, root);
    if (!body || !progress) return;

    var rect = body.getBoundingClientRect();
    var pageY = window.pageYOffset || document.documentElement.scrollTop || 0;
    var viewportBottom = pageY + (window.innerHeight || document.documentElement.clientHeight || 0);
    var bodyTop = rect.top + pageY;
    var bodyHeight = Math.max(body.scrollHeight, rect.height, 1);
    var value = Math.max(0, Math.min(100, ((viewportBottom - bodyTop) / bodyHeight) * 100));

    progress.style.transform = "scaleX(" + (value / 100).toFixed(4) + ")";
    progress.setAttribute("aria-valuenow", String(Math.round(value)));
  }

  function findRteSource(sourceCandidates){
    sourceCandidates = Array.isArray(sourceCandidates) ? sourceCandidates : [sourceCandidates];

    for (var i = 0; i < sourceCandidates.length; i++) {
      var candidate = sourceCandidates[i];
      var source = "";

      if (!candidate) continue;
      if (typeof candidate === "string") {
        source = qs(candidate);
      } else if (candidate.nodeType === 1) {
        source = candidate;
      }

      if (!source) continue;
      if (source.classList && source.classList.contains("w-richtext")) return source;

      var rte = qs(".w-richtext", source);
      if (rte) return rte;
    }

    return null;
  }

      var words = body ? ((body.innerText || body.textContent || "").match(/\S+/g) || []).length : 0;

  function planInfo(raw){
    var value = clean(raw);
    if (!value) return null;

      .split(/[?#]/)[0]
      .replace(/^https?:\/\/[^/]+/i, "")
      .replace(/^\/?training-plans\//i, "")
      .replace(/^\/+|\/+$/g, "");

    var labels = {
      "6-week-advanced-hypertrophy-training-plan": "6-Week Advanced Hypertrophy Plan",
      "6-week-strength-hypertrophy-training-plan": "6-Week Strength + Hypertrophy Plan",

      "full-body-dynamic-warm-up-routine": "Full-Body Dynamic Warm-Up Routine",
      "full-body-static-stretching-routine": "Full-Body Static Stretching Routine"
    };

  function youtubeVimeoEmbed(raw){
    var href = validUrl(raw);
    if (!href) return "";
    try {
      var url = new URL(href);
      var host = url.hostname.replace(/^www\./, "");
      var id = "";

      if (host === "youtu.be") {
        id = url.pathname.replace(/^\/+/, "").split("/")[0];
        return id ? "https://www.youtube.com/embed/" + encodeURIComponent(id) : "";
      }

      if (host === "youtube.com" || host === "m.youtube.com") {
        id = url.searchParams.get("v") || "";
        if (!id && url.pathname.indexOf("/embed/") === 0) id = url.pathname.split("/embed/")[1].split("/")[0];
        return id ? "https://www.youtube.com/embed/" + encodeURIComponent(id) : "";
      }

      if (host === "vimeo.com" || host === "player.vimeo.com") {
        id = (url.pathname.match(/\d+/) || [])[0] || "";
        return id ? "https://player.vimeo.com/video/" + encodeURIComponent(id) : "";
      }
    } catch(e) {}
    return "";
  }

  function setupScrollDepthTracking(){
    if (scrollTrackingStarted) return;
    if (!root || root.getAttribute("data-nf-body-ready") !== "true") return;
    var body = qs(BODY_SLOT_SELECTOR, root);
    if (!body || !hasSlotContent(body)) return;
    scrollTrackingStarted = true;

    var thresholds = [25, 50, 75, 100];
    var eventNames = {
      25: "nf_blog_scroll_25",
      50: "nf_blog_scroll_50",
      75: "nf_blog_scroll_75",
      100: "nf_blog_scroll_100"
    };
    var fired = {};

    function onScrollDepth(){
      ticking = false;
      updateReadingProgress();
      updateTocCurrent();

      var rect = body.getBoundingClientRect();
      var pageY = window.pageYOffset || document.documentElement.scrollTop || 0;
      var viewportBottom = pageY + (window.innerHeight || document.documentElement.clientHeight || 0);
      var bodyTop = rect.top + pageY;
      var bodyHeight = Math.max(body.scrollHeight, rect.height, 1);
      var progress = Math.max(0, Math.min(100, ((viewportBottom - bodyTop) / bodyHeight) * 100));

      thresholds.forEach(function(depth){
        if (fired[depth] || progress < depth) return;
        var payload = blogPayload();
        payload.scroll_depth = depth;
        trackOnce("blog-article-scroll-" + (data("slug") || location.pathname) + "-" + depth, eventNames[depth], payload);
        fired[depth] = true;
      });

    }

    function scheduleScrollDepth(){
      if (ticking) return;
      ticking = true;
      (window.requestAnimationFrame || function(fn){ return window.setTimeout(fn, 16); })(onScrollDepth);
    }

    window.addEventListener("scroll", scheduleScrollDepth, { passive: true });
    window.addEventListener("resize", scheduleScrollDepth);
    scheduleScrollDepth();
  }

  function tryCloneBody(finalAttempt){
    if (bodyBridgeDone && root.getAttribute("data-nf-body-ready") === "true") return true;

    var slot = qs(BODY_SLOT_SELECTOR, root);
    if (!slot) {
      if (finalAttempt) {
        setBodyMissing("slot-missing");
        bodyBridgeDone = true;

      buildToc();
      setupScrollDepthTracking();
      maybeTrackView();

    if (!source) {
      if (finalAttempt) {
        setBodyMissing("source-not-found");
        bodyBridgeDone = true;
        maybeTrackView();
      }
      return false;
    }

    if (!hasRteContent(source)) {
      if (finalAttempt) {
        setBodyMissing("source-empty");
        bodyBridgeDone = true;
        maybeTrackView();
      }
      return false;
    }

    if (!cloneRteIntoSlot(source, slot)) {
      if (finalAttempt) {
        setBodyMissing("source-empty");
        bodyBridgeDone = true;
        maybeTrackView();
      }
      return false;
    }

    setBodyReady();
    bodyBridgeDone = true;
    updateReadTime(true);
    buildToc();
    setupScrollDepthTracking();
    maybeTrackView();
    return true;
  }

  function tryCloneTakeaways(finalAttempt){

    var slot = qs(TAKEAWAYS_SLOT_SELECTOR, root);

        takeawaysBridgeDone = true;
      }
      return false;
    }

    if (!cloneRteIntoSlot(source, slot)) {
      if (finalAttempt) {
        setTakeawaysMissing();
        takeawaysBridgeDone = true;
      }
      return false;
    }

    setTakeawaysReady();
    if (wrap) show(wrap);
    takeawaysBridgeDone = true;
    return true;
  }

  function stopBridgeBoot(){
    if (bridgeTimer) {
      window.clearTimeout(bridgeTimer);
      bridgeTimer = 0;
    }
    if (bridgeObserver) {
      try { bridgeObserver.disconnect(); } catch(e) {}
      bridgeObserver = null;
    }
  }

  function runBridgeAttempt(finalAttempt){
    if (!root) return;

    var bodyOk = tryCloneBody(finalAttempt);
    var takeawaysOk = tryCloneTakeaways(finalAttempt);
    var done = bodyBridgeDone && takeawaysBridgeDone;

    if (bodyOk && takeawaysOk) done = true;
    if (done || finalAttempt) {
      if (!bodyBridgeDone) tryCloneBody(true);
      if (!takeawaysBridgeDone) tryCloneTakeaways(true);
      hideEmptyContainers();
      stopBridgeBoot();
      return;
    }

    scheduleBridgeAttempt();
  }

  function scheduleBridgeAttempt(){
    if (bridgeTimer) return;
    if (bridgeAttempts >= MAX_BRIDGE_ATTEMPTS) {
      runBridgeAttempt(true);
      return;
    }

    bridgeTimer = window.setTimeout(function(){
      bridgeTimer = 0;
      bridgeAttempts += 1;
      runBridgeAttempt(bridgeAttempts >= MAX_BRIDGE_ATTEMPTS);
    }, BRIDGE_RETRY_DELAY);
  }

  function observeBridgeSources(){
    if (!("MutationObserver" in window) || bridgeObserver || !document.body) return;

    bridgeObserver = new MutationObserver(function(){
      if (bodyBridgeDone && takeawaysBridgeDone) {
        stopBridgeBoot();
        return;
      }
      runBridgeAttempt(false);
    });

    try {
      bridgeObserver.observe(document.body, {
        childList: true,
        subtree: true,
        attributes: true,
        attributeFilter: ["class", "hidden", "style", "data-rte-body", "data-rte-takeaways"]
      });
    } catch(e) {
      bridgeObserver = null;
    }
  }

  function startBridgeBoot(){
    bridgeAttempts = 1;
    observeBridgeSources();
    runBridgeAttempt(false);
    scheduleBridgeAttempt();
  }

  function init(){
    if (booted) return;

    root = qs(ROOT_SELECTOR);
    if (!root) {
      scheduleRootBoot();
      return;
    }

    if (root.__nfBlogPostReady) return;
    root.__nfBlogPostReady = true;
    booted = true;

    if (!document.querySelector('main, [role="main"]')) {
      root.setAttribute("role", "main");
    }
    if (!root.hasAttribute("tabindex")) {
      root.setAttribute("tabindex", "-1");
    }

    tuneStructuredData();
    hideEmptyNodes();
    renderTags();
    renderRelatedPlans();
    updateReadTime(false);
    hideDuplicateUpdatedDate();
    setupVideo();
    setupAudio();
    setupCta();
    normalizeArticleSemantics();
    setupAuthorLink();
    setupArticleNav();
    hideEmptyContainers();
    maybeTrackView();
    startBridgeBoot();
  }

  function scheduleRootBoot(){
    if (booted || bridgeTimer) return;
    if (bridgeAttempts >= MAX_BRIDGE_ATTEMPTS) return;

    bridgeTimer = window.setTimeout(function(){
      bridgeTimer = 0;
      bridgeAttempts += 1;
      init();
    }, BRIDGE_RETRY_DELAY);
  }

  function onReady(){
    init();
  }

  init();

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", onReady, { once: true });
  } else {
    onReady();
  }
})();
